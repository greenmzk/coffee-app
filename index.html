    function handleAction(action){

      switch(action){

        case "start":

          localStorage.setItem(WELCOME_KEY,"1");state.screen="home";render();break;

        case "new-recipe":

          state.editingId=null;state.detailId=null;state.backTarget=state.screen==="calendar"?"calendar":"home";

          state.draft=freshDraft();state.screen="editor";render();break;

        case "back":handleBack();break;

        case "save-recipe":saveRecipe();break;

        case "pick-date":openDateModal();break;

        case "add-pour":syncVisibleFields();state.draft.pours.push({water:"",time:""});renderEditor();break;

        case "edit-current":editRecipe(state.detailId);break;

        case "delete-current":deleteCurrent();break;

        case "add-custom-equipment":openCustomEquipmentForm();break;

        case "premium":openPremiumModal();break;

        case "settings":

          state.settingsOrigin=state.screen==="profile"?"home":state.screen;

          state.screen="profile";render();break;

        case "export":exportData();break;

        case "import":document.getElementById("importFile")?.click();break;

        case "clear-recipes":

          confirmBox(t("deleteAllTitle"),t("deleteAllMessage"),()=>{

            state.recipes=[];saveRecipes();renderProfile();toast(t("recipesDeleted"));

          });break;

      }

    }


    function handleBack(){

      if(state.screen==="detail"){state.screen=state.backTarget||"home";render();return}

      state.screen="home";render();

    }

    function saveRecipe(){

      const roastInput=app.querySelector("[data-roast-date]");

      if(roastInput){

        const parsed=parseRoastDate(roastInput.value);

        if(parsed===null)return toast(t("invalidRoastDate"));

        state.draft.roastDate=parsed;

      }

      syncVisibleFields();const d=state.draft;

      if(!d.date)return toast(t("dateRequired"));

      if(!String(d.beanOrigin||"").trim())d.beanOrigin=t("unnamedCoffee");


      if(d.milk==="__custom__"){

        const name=String(d.customMilk||"").trim();if(!name)return toast(t("milkRequired"));

        if(!customMilks.includes(name)){customMilks.push(name);saveCustomMilks()}

        d.milk=name;d.customMilk="";

      }

      if(d.beverageType!=="Ice")d.iceWeight="";


      const recipe=JSON.parse(JSON.stringify(d));

      recipe.beanWeight=toNumberOrString(recipe.beanWeight);

      recipe.waterTemp=toNumberOrString(recipe.waterTemp);

      recipe.bloomWater=toNumberOrString(recipe.bloomWater);

      recipe.iceWeight=toNumberOrString(recipe.iceWeight);

      recipe.updatedAt=new Date().toISOString();


      const wasEditing=!!state.editingId;

      if(wasEditing){

        recipe.id=state.editingId;

        const idx=state.recipes.findIndex(r=>r.id===state.editingId);

        if(idx>=0){recipe.createdAt=state.recipes[idx].createdAt||recipe.updatedAt;state.recipes[idx]=recipe}

        else state.recipes.push(recipe);

      }else{

        recipe.id=globalThis.crypto?.randomUUID?.()||("r_"+Date.now()+"_"+Math.random().toString(16).slice(2));

        recipe.createdAt=new Date().toISOString();state.recipes.push(recipe);

      }

      saveRecipes();state.detailId=recipe.id;state.editingId=null;state.screen="detail";renderDetail();toast(wasEditing?t("updated"):t("saved"));

    }

    function exportData(){

      const data={app:"Brew Day",version:4,exportedAt:new Date().toISOString(),language:state.lang,temperatureUnit:state.tempUnit,recipes:state.recipes,customEquipment,customMilks};

      const blob=new Blob([JSON.stringify(data,null,2)],{type:"application/json"});

    }

    async function importData(e){

      const file=e.target.files?.[0];if(!file)return;

      try{

        const text=await file.text(),data=JSON.parse(text);

        const recipes=Array.isArray(data)?data:data.recipes;

        if(!Array.isArray(recipes))throw new Error("invalid");

        const importedCustom=Array.isArray(data.customEquipment)?data.customEquipment:[];

        const importedMilks=Array.isArray(data.customMilks)?data.customMilks:[];

        confirmBox(t("importTitle"),t("importMessage",recipes.length),()=>{

          state.recipes=recipes.map(normalizeRecipe);

          customEquipment=importedCustom.map((x,i)=>({...x,source:"custom",id:x.id||("ce_import_"+i)}));

          customMilks=[...new Set(importedMilks.filter(Boolean))];

          if(data.language==="ja"||data.language==="en"){state.lang=data.language;localStorage.setItem(LANG_KEY,state.lang)}

          if(data.temperatureUnit==="C"||data.temperatureUnit==="F"){state.tempUnit=data.temperatureUnit;localStorage.setItem(TEMP_UNIT_KEY,state.tempUnit)}

          saveRecipes();saveCustomEquipment();saveCustomMilks();renderProfile();toast(t("importDone"));

        });

      }catch(_){toast(t("importError"))}

      e.target.value="";

    }
