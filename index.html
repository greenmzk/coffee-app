  function renderCalendar() {

    const now = new Date();

    const year = now.getFullYear();

    const month = now.getMonth();


    document.getElementById("calendarTitle").textContent =

      `${year}年 ${month + 1}月`;


    const firstDay = new Date(year, month, 1).getDay();

    const days = new Date(year, month + 1, 0).getDate();


    const recipeDays = new Set(

      recipes

        .filter(recipe => {

          const d = new Date(recipe.date + "T12:00:00");

          return d.getFullYear() === year && d.getMonth() === month;

        })

        .map(recipe => new Date(recipe.date + "T12:00:00").getDate())

    );


    const grid = document.getElementById("calendarGrid");


    let html =

      ["日", "月", "火", "水", "木", "金", "土"]

        .map(day => `<span class="dow">${day}</span>`)

        .join("");


    for (let i = 0; i < firstDay; i++) {

      html += `<span></span>`;

    }


    for (let day = 1; day <= days; day++) {

      html += `

        <span class="${recipeDays.has(day) ? "has-recipe" : ""}">

          ${day}

        </span>

      `;

    }


    grid.innerHTML = html;


    const currentMonthRecipes = recipes.filter(recipe => {

      const d = new Date(recipe.date + "T12:00:00");

      return d.getFullYear() === year && d.getMonth() === month;

    });


    document.getElementById("calendarRecipes").innerHTML =

      currentMonthRecipes.length

        ? currentMonthRecipes.map(recipe => `

            <article

              class="recipe-card"

              data-calendar-recipe="${recipe.id}"

            >

              <div class="recipe-date">

                <span class="day">

                  ${new Date(recipe.date + "T12:00:00").getDate()}

                </span>

              </div>


              <div class="recipe-info">

                <h3>${escapeHTML(recipe.beanName)}</h3>

                <div class="beans">

                  ${beanHTML(recipe.rating || 0)}

                </div>

              </div>


              <span class="chevron">›</span>

            </article>

          `).join("")

        : `<div class="empty-card"><p>今月の記録はありません。</p></div>`;


    document.querySelectorAll("[data-calendar-recipe]").forEach(card => {

      card.addEventListener("click", () => {

        openRecipeDetail(card.dataset.calendarRecipe);

      });

    });

  }

  function renderProfile() {

    document.getElementById("recipeCount").textContent = recipes.length;

    document.getElementById("equipmentCount").textContent = equipment.length;


    const rated = recipes.filter(recipe => recipe.rating > 0);


    const average = rated.length

          rated.reduce((sum, recipe) => sum + recipe.rating, 0) /

          rated.length

        ).toFixed(1)

      : "—";


    document.getElementById("averageRating").textContent = average;

  }
